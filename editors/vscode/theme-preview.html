; Monte-Carlo portfolio VaR
(module portfolio)

(import std.math   '[sqrt log exp])
(import std.tensor '[matrix dot])

(defn simulate-paths
  [S0 mu sigma T N]
  (let [dt   (/ T N)
        drift (* (- mu (/ (* sigma sigma) 2.0)) dt)]
    (for i in (range N)
      (* S0 (exp (+ drift
                       (* sigma (sqrt dt)
                          (randn))))))))

(defn var-95 [returns]
  (let [sorted (sort returns)
        idx    (floor (* 0.05 (len sorted)))]
    (nth sorted idx)))

(try
  (let [paths (simulate-paths 100.0 0.08 0.2 1.0 252)]
    (println "VaR 95%:" (var-95 paths)))
  (catch e (error "sim failed: " e)))

(module nqueens)

; Solve N-Queens via logic unification
(defrel safe [?q ?rest ?d]
  (conde
    [(== ?rest '())]
    [(fresh [?h ?t]
       (== ?rest (cons ?h ?t))
       (=/= ?q ?h)
       (=/= ?q (+ ?h ?d))
       (=/= ?q (- ?h ?d))
       (safe ?q ?t (+ ?d 1)))]))

(run* [?q] (nqueens 8 ?q))
